# LLM Text Watermarking: run commands

# 1. Install dependencies
pip install -r llm_watermarking/requirements.txt

# 2. Generate baseline outputs used for comparison
powershell -ExecutionPolicy Bypass -File run_baseline.ps1 -Dataset c4 -NumSamples 100 -MaxTokens 150 -OutputDir outputs

# 3. Generate final Undetectable watermark runs
powershell -ExecutionPolicy Bypass -File run_pipeline.ps1 -Schemes Undetectable -Lambdas 2.0,4.0,5.0,6.0 -NumSamples 100 -MaxTokens 150 -OutputDir outputs

# 4. Generate final PRC runs used in the report
python -m llm_watermarking.main --watermark PRC --dataset c4 --max-tokens 150 --num-samples 100 --suffix _c4_s100_t150 --output-dir outputs --no-plots --prc-n 2048 --prc-g 128 --prc-t 1 --prc-r 2028 --prc-eta 0.005 --prc-zeta 0.045
python -m llm_watermarking.main --watermark PRC --dataset c4 --max-tokens 150 --num-samples 5 --suffix _c4_s5_t150_n512 --output-dir outputs --no-plots --prc-n 512 --prc-g 32 --prc-t 1 --prc-r 500 --prc-eta 0.005 --prc-zeta 0.045
python -m llm_watermarking.main --watermark PRC --dataset c4 --max-tokens 300 --num-samples 100 --suffix _c4_s100_t300_n2048 --output-dir outputs --no-plots --prc-n 2048 --prc-g 128 --prc-t 1 --prc-r 2028 --prc-eta 0.005 --prc-zeta 0.045

# 5. Evaluate detectability and quality
python -m llm_watermarking.evaluate_experiment --baseline outputs/baseline_results.jsonl --watermarked outputs/prc_results_c4_s100_t150.jsonl
python -m llm_watermarking.evaluate_experiment --baseline outputs/baseline_results.jsonl --watermarked outputs/prc_results_c4_s5_t150_n512.jsonl
python -m llm_watermarking.evaluate_experiment --baseline outputs/baseline_results.jsonl --watermarked outputs/prc_results_c4_s100_t300_n2048.jsonl
python -m llm_watermarking.evaluate_experiment --baseline outputs/baseline_results.jsonl --watermarked outputs/undetectable_results_lam2.0.jsonl
python -m llm_watermarking.evaluate_experiment --baseline outputs/baseline_results.jsonl --watermarked outputs/undetectable_results_lam4.0.jsonl
python -m llm_watermarking.evaluate_experiment --baseline outputs/baseline_results.jsonl --watermarked outputs/undetectable_results_lam5.0.jsonl
python -m llm_watermarking.evaluate_experiment --baseline outputs/baseline_results.jsonl --watermarked outputs/undetectable_results_lam6.0.jsonl

# 6. Evaluate robustness under random substitutions
python -m llm_watermarking.evaluate_robustness --results outputs/prc_results_c4_s100_t150.jsonl --output outputs/table3_robustness_prc_c4_s100_t150.csv
python -m llm_watermarking.evaluate_robustness --results outputs/undetectable_results_lam5.0.jsonl --output outputs/robustness_results_undetectable_lam5.0.csv

# 7. Rebuild plots from existing CSVs/results when needed
python -m llm_watermarking.visualization
